40  /**
41   * Validates and snapshots local mail resources once per run. No network resolver is used.
42   * Files are kept in memory so later filesystem changes cannot change a validated message.
43   */
44  public final class MailResources {
45      private final Path directory;
46      private final Map<String, Asset> references = new HashMap<>();
47      private final Map<Path, Asset> files = new HashMap<>();
48      private final List<Asset> attachments = new ArrayList<>();
49  
50      /**
51       * Load configured, comma-separated attachments relative to the real template directory.
52       * @param configuration template directory and attachment references.
53       * @throws IOException if a configured file is inaccessible or outside the directory.
54       */
55      public MailResources(MailConfig configuration) throws IOException {
56          directory = Path.of(configuration.getTemplatePath()).toRealPath();
57          if (!Files.isDirectory(directory)) {
58              throw new IOException("Template directory is not a directory: " + directory);
59          }
60          Map<Path, Asset> unique = new LinkedHashMap<>();
61          for (String reference : Objects.requireNonNullElse(configuration.getAttachments(), "").split(",")) {
62              if (!reference.isBlank()) {
63                  Asset asset = load(reference.trim());
64                  unique.putIfAbsent(asset.path(), asset);
65              }
66          }
67          attachments.addAll(unique.values());
68      }
69  
70      /**
71       * Validate every inline image in a rendered message before the first delivery.
72       * @param template rendered message bodies.
73       * @throws IOException for missing, external, unsupported or escaping image references.
74       */
75      public void validate(MailTemplate template) throws IOException {
76          Document document = parse(template.getHtml());
77          if (!document.select("[srcset]").isEmpty()) {
78              throw new IOException("srcset is not supported; use local img src references");
79          }
80          for (Element image : document.select("img")) {
81              Asset asset = load(image.attr("src"));
82              if (!asset.contentType().startsWith("image/")) {
83                  throw new IOException("Inline resource is not an image: " + image.attr("src"));
84              }
85          }
86      }
87  
88      /**
89       * Embed local images using Content-IDs and add ordinary downloadable attachments.
90       * @param email configured outgoing message.
91       * @param template rendered HTML and plain-text alternatives.
92       * @throws EmailException if resource validation or MIME construction fails.
93       */
94      public void applyTo(HtmlEmail email, MailTemplate template) throws EmailException {
95          try {
96              validate(template);
97              Document document = parse(template.getHtml());
98              Map<Path, String> contentIds = new HashMap<>();
99              for (Element image : document.select("img")) {
100                 Asset asset = load(image.attr("src"));
101                 String cid = contentIds.get(asset.path());
102                 if (cid == null) {
103                     cid = email.embed(asset.dataSource(), directory.relativize(asset.path()).toString());
104                     contentIds.put(asset.path(), cid);
105                 }
106                 image.attr("src", "cid:" + cid);
107             }
108             email.setHtmlMsg(contentIds.isEmpty() ? template.getHtml() : document.outerHtml());
109             email.setTextMsg(template.getPlaintext());
110             for (Asset asset : attachments) {
111                 email.attach(asset.dataSource(), asset.filename(), "", EmailAttachment.ATTACHMENT);
112             }
113         } catch (IOException e) {
114             throw new EmailException("Cannot prepare local mail resources", e);
115         }
116     }
117 
118     private Document parse(String html) {
119         Document document = Jsoup.parse(html);
120         document.outputSettings().prettyPrint(false);
121         return document;
122     }
123 
124     private Asset load(String reference) throws IOException {
125         Asset cached = references.get(reference);
126         if (cached != null) {
127             return cached;
128         }
129         Path relative;
130         try {
131             URI uri = new URI(reference.replace(" ", "%20"));
132             if (reference.isBlank() || reference.contains("\\") || uri.isAbsolute()
133                     || uri.getRawAuthority() != null || uri.getRawQuery() != null || uri.getRawFragment() != null) {
134                 throw new IOException("Only relative local resource paths are allowed: " + reference);
135             }
136             String path = uri.getPath();
137             if (path == null) {
138                 throw new IOException("Resource path is missing: " + reference);
139             }
140             relative = Path.of(path);
141         } catch (URISyntaxException | IllegalArgumentException e) {
142             throw new IOException("Invalid local resource path: " + reference, e);
143         }
144         if (relative.isAbsolute()) {
145             throw new IOException("Absolute resource paths are not allowed: " + reference);
146         }
147         Path file = directory.resolve(relative).toRealPath();
148         if (!file.startsWith(directory) || !Files.isRegularFile(file) || !Files.isReadable(file)) {
149             throw new IOException("Resource must be a readable file inside templatepath: " + reference);
150         }
151         Asset asset = files.get(file);
152         if (asset == null) {
153             Path filename = file.getFileName();
154             if (filename == null) {
155                 throw new IOException("Resource filename is missing: " + reference);
156             }
157             String contentType = URLConnection.guessContentTypeFromName(filename.toString());
158             if (contentType == null) {
159                 contentType = "application/octet-stream";
160             }
161             asset = new Asset(file, filename.toString(), Files.readAllBytes(file), contentType);
162             files.put(file, asset);
163         }
164         references.put(reference, asset);
165         return asset;
166     }
167 
168     private record Asset(Path path, String filename, byte[] bytes, String contentType) {
169         ByteArrayDataSource dataSource() {
170             ByteArrayDataSource source = new ByteArrayDataSource(bytes, contentType);
171             source.setName(filename);
172             return source;
173         }
174     }

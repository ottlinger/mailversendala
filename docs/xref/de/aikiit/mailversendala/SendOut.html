40  /**
41   * This class performs the actual mail sending and encapsulates all the technicalities to create and send out an email.
42   */
43  public class SendOut {
44      private static final Logger LOG =
45              LogManager.getLogger(SendOut.class);
46  
47      // is transformed into User-Agent-Header
48      private static final String SENT_BY = "X-Mailer";
49      private static final String SENT_BY_ME = "Mailversendala - https://github.com/ottlinger/mailversendala";
50  
51      private final Email email;
52      private final HtmlEmail htmlEmail;
53      private final MailConfig mailConfig;
54      private final Mailing recipient;
55  
56      /**
57       * Create mail based on the given mailing recipient.
58       * @param recipient mailing contents and parameters.
59       */
60      public SendOut(Mailing recipient) {
61          this(recipient, new MailConfig());
62      }
63  
64      /**
65       * Create a sender using the same configuration as the mailing run.
66       * @param recipient mailing recipient.
67       * @param mailConfig SMTP and message configuration.
68       */
69      public SendOut(Mailing recipient, MailConfig mailConfig) {
70          this(recipient, mailConfig, new SimpleEmail(), new HtmlEmail());
71      }
72  
73      /**
74       * Configure supplied mail clients, allowing tests to use isolated mail sessions.
75       */
76      SendOut(Mailing recipient, MailConfig mailConfig, Email email, HtmlEmail htmlEmail) {
77          this.recipient = recipient;
78          this.mailConfig = mailConfig;
79          this.email = email;
80          this.htmlEmail = htmlEmail;
81  
82          if (!mailConfig.sendOutMails()) {
83              LOG.warn("Running in DEMO mode, thus no mails will be sent out.");
84          }
85  
86          LOG.info("Initialized authentication.");
87          DefaultAuthenticator authenticator = new DefaultAuthenticator(mailConfig.getUsername(), mailConfig.getPassword());
88  
89          int port = mailConfig.getPort();
90  
91          email.setHostName(mailConfig.getHost());
92          email.setSmtpPort(port);
93          email.setAuthenticator(authenticator);
94          email.setSSLCheckServerIdentity(true);
95          if (465 == port) {
96              LOG.info("Using SSL.");
97              email.setSSLOnConnect(true);
98          } else {
99              LOG.info("Using TLS.");
100             email.setStartTLSEnabled(true);
101             email.setStartTLSRequired(true);
102         }
103         email.setBounceAddress(mailConfig.getFrom());
104         email.addHeader(SENT_BY, SENT_BY_ME);
105         LOG.info("Simple email initialized.");
106 
107         htmlEmail.setHostName(mailConfig.getHost());
108         htmlEmail.setSmtpPort(port);
109         htmlEmail.setSSLCheckServerIdentity(true);
110         if (465 == port) {
111             LOG.info("Using SSL.");
112             htmlEmail.setSSLOnConnect(true);
113         } else {
114             LOG.info("Using TLS.");
115             htmlEmail.setStartTLSEnabled(true);
116             htmlEmail.setStartTLSRequired(true);
117         }
118         htmlEmail.setBounceAddress(mailConfig.getFrom());
119         htmlEmail.setAuthenticator(authenticator);
120         htmlEmail.addHeader(SENT_BY, SENT_BY_ME);
121         LOG.info("HTML and text email initialized.");
122     }
123 
124     /**
125      * Test method to verify the actual mail creation and send out.
126      * @param args allows sending an email to a given mail address. One parameter is required.
127      * @throws Exception in case of errors.
128      */
129     public static void main(String... args) throws Exception {
130         if (args == null || args.length < 1 || Strings.isNullOrEmpty(args[0])) {
131             LOG.error("Please call this method with a mail address to send to.");
132         } else {
133             String email = args[0];
134             LOG.warn("Init: Using mail address from runtime parameter: {}", email);
135 
136             Mailing mailing = Mailing.builder().email(email).firstname("Your name").surname("Is my name").build();
137             MailConfig configuration = new MailConfig();
138             VelocityMailTemplate templates = new VelocityMailTemplate(configuration);
139             SendOut sendOut = new SendOut(mailing, configuration);
140             sendOut.send(templates.render(mailing));
141             LOG.info("Send HTML and plain-text message: DONE");
142         }
143     }
144 
145     /**
146      * Send a rendered message with HTML and plain-text alternatives in UTF-8.
147      * @param template rendered message bodies.
148      * @throws EmailException if message construction or delivery fails.
149      */
150     public void send(MailTemplate template) throws EmailException {
151         try {
152             send(template, new MailResources(mailConfig));
153         } catch (IOException e) {
154             throw new EmailException("Cannot load mail resources", e);
155         }
156     }
157 
158     /** Send using the resources validated and snapshotted for the entire run. */
159     void send(MailTemplate template, MailResources resources) throws EmailException {
160         htmlEmail.setCharset(Charsets.UTF_8.name());
161         htmlEmail.setFrom(mailConfig.getFrom());
162         htmlEmail.addTo(recipient.getEmail());
163         htmlEmail.setSubject(mailConfig.getSubject());
164         resources.applyTo(htmlEmail, template);
165         if (mailConfig.sendOutMails()) {
166             htmlEmail.send();
167         }
168     }
169 
170     /**
171      * Send out an email that contains only text contents.
172      * @throws EmailException in case of errors.
173      */
174     public void send() throws EmailException {
175         email.setFrom(mailConfig.getFrom());
176         email.addTo(this.recipient.getEmail());
177 
178         email.setCharset(Charsets.UTF_8.name());
179 
180         email.setSubject(mailConfig.getSubject() + " " + new Date());
181         email.setMsg("This is a test mail from Mailversendala... :-)");
182         if (mailConfig.sendOutMails()) {
183             email.send();
184         }
185     }
186 
187     /**
188      * Send out an email with plain text and HTML contents.
189      *
190      * @throws EmailException in case of mail-related errors.
191      * @throws MalformedURLException in case of problem while generating the HTML-part of.
192      */
193     public void sendComplex() throws EmailException, MalformedURLException {
194         sendComplex(new URL("https://www.apache.org/images/asf_logo_wide.gif"));
195     }
196 
197     /**
198      * Send HTML mail using the supplied inline image, without requiring a remote resource.
199      */
200     void sendComplex(URL imageUrl) throws EmailException {
201         htmlEmail.addTo(this.recipient.getEmail(), this.recipient.getFirstname() + " " + this.recipient.getSurname());
202         htmlEmail.setFrom(mailConfig.getFrom(), "Me");
203         htmlEmail.setSubject("HTML" + mailConfig.getSubject() + " " + new Date());
204 
205         // embed the image and get the content id
206         String cid = htmlEmail.embed(imageUrl, "ASF logo");
207 
208         // set the html message
209         htmlEmail.setHtmlMsg("<html>The ASF logo - <img src=\"cid:" + cid + "\"></html>");
210 
211         // set the alternative message
212         htmlEmail.setTextMsg("Your email client does not support HTML messages - thus no ASF logo visible.");
213 
214         if (mailConfig.sendOutMails()) {
215             htmlEmail.send();
216         }
217     }

34  /**
35   * This class performs the actual mail sending and encapsulates all the technicalities to create and send out an email.
36   */
37  public class SendOut {
38      private static final Logger LOG =
39              LogManager.getLogger(SendOut.class);
40  
41      // is transformed into User-Agent-Header
42      private static final String SENT_BY = "X-Mailer";
43      private static final String SENT_BY_ME = "Mailversendala - https://github.com/ottlinger/mailversendala";
44  
45      private final Email email;
46      private final HtmlEmail htmlEmail;
47      private final MailConfig mailConfig;
48      private final Mailing recipient;
49  
50      /**
51       * Create mail based on the given mailing recipient.
52       * @param recipient mailing contents and parameters.
53       */
54      public SendOut(Mailing recipient) {
55          this(recipient, new MailConfig());
56      }
57  
58      /**
59       * Create a sender using the same configuration as the mailing run.
60       * @param recipient mailing recipient.
61       * @param mailConfig SMTP and message configuration.
62       */
63      public SendOut(Mailing recipient, MailConfig mailConfig) {
64          this(recipient, mailConfig, new SimpleEmail(), new HtmlEmail());
65      }
66  
67      /**
68       * Configure supplied mail clients, allowing tests to use isolated mail sessions.
69       */
70      SendOut(Mailing recipient, MailConfig mailConfig, Email email, HtmlEmail htmlEmail) {
71          this.recipient = recipient;
72          this.mailConfig = mailConfig;
73          this.email = email;
74          this.htmlEmail = htmlEmail;
75  
76          if (!mailConfig.sendOutMails()) {
77              LOG.warn("Running in DEMO mode, thus no mails will be sent out.");
78          }
79  
80          LOG.info("Initialized authentication.");
81          DefaultAuthenticator authenticator = new DefaultAuthenticator(mailConfig.getUsername(), mailConfig.getPassword());
82  
83          int port = mailConfig.getPort();
84  
85          email.setHostName(mailConfig.getHost());
86          email.setSmtpPort(port);
87          email.setAuthenticator(authenticator);
88          email.setSSLCheckServerIdentity(true);
89          if (465 == port) {
90              LOG.info("Using SSL.");
91              email.setSSLOnConnect(true);
92          } else {
93              LOG.info("Using TLS.");
94              email.setStartTLSEnabled(true);
95              email.setStartTLSRequired(true);
96          }
97          email.setBounceAddress(mailConfig.getFrom());
98          email.addHeader(SENT_BY, SENT_BY_ME);
99          LOG.info("Simple email initialized.");
100 
101         htmlEmail.setHostName(mailConfig.getHost());
102         htmlEmail.setSmtpPort(port);
103         htmlEmail.setSSLCheckServerIdentity(true);
104         if (465 == port) {
105             LOG.info("Using SSL.");
106             htmlEmail.setSSLOnConnect(true);
107         } else {
108             LOG.info("Using TLS.");
109             htmlEmail.setStartTLSEnabled(true);
110             htmlEmail.setStartTLSRequired(true);
111         }
112         htmlEmail.setBounceAddress(mailConfig.getFrom());
113         htmlEmail.setAuthenticator(authenticator);
114         htmlEmail.addHeader(SENT_BY, SENT_BY_ME);
115         LOG.info("HTML and text email initialized.");
116     }
117 
118     /**
119      * Test method to verify the actual mail creation and send out.
120      * @param args allows sending an email to a given mail address. One parameter is required.
121      * @throws Exception in case of errors.
122      */
123     public static void main(String... args) throws Exception {
124         if (args == null || args.length < 1 || Strings.isNullOrEmpty(args[0])) {
125             LOG.error("Please call this method with a mail address to send to.");
126         } else {
127             String email = args[0];
128             LOG.warn("Init: Using mail address from runtime parameter: {}", email);
129 
130             Mailing mailing = Mailing.builder().email(email).firstname("Your name").surname("Is my name").build();
131             MailConfig configuration = new MailConfig();
132             VelocityMailTemplate templates = new VelocityMailTemplate(configuration);
133             SendOut sendOut = new SendOut(mailing, configuration);
134             sendOut.send(templates.render(mailing));
135             LOG.info("Send HTML and plain-text message: DONE");
136         }
137     }
138 
139     /**
140      * Send a rendered message with HTML and plain-text alternatives in UTF-8.
141      * @param template rendered message bodies.
142      * @throws EmailException if message construction or delivery fails.
143      */
144     public void send(MailTemplate template) throws EmailException {
145         htmlEmail.setCharset(Charsets.UTF_8.name());
146         htmlEmail.setFrom(mailConfig.getFrom());
147         htmlEmail.addTo(recipient.getEmail());
148         htmlEmail.setSubject(mailConfig.getSubject());
149         htmlEmail.setHtmlMsg(template.getHtml());
150         htmlEmail.setTextMsg(template.getPlaintext());
151         if (mailConfig.sendOutMails()) {
152             htmlEmail.send();
153         }
154     }
155 
156     /**
157      * Send out an email that contains only text contents.
158      * @throws EmailException in case of errors.
159      */
160     public void send() throws EmailException {
161         email.setFrom(mailConfig.getFrom());
162         email.addTo(this.recipient.getEmail());
163 
164         email.setCharset(Charsets.UTF_8.name());
165 
166         email.setSubject(mailConfig.getSubject() + " " + new Date());
167         email.setMsg("This is a test mail from Mailversendala... :-)");
168         if (mailConfig.sendOutMails()) {
169             email.send();
170         }
171     }
172 
173     /**
174      * Send out an email with plain text and HTML contents.
175      *
176      * @throws EmailException in case of mail-related errors.
177      * @throws MalformedURLException in case of problem while generating the HTML-part of.
178      */
179     public void sendComplex() throws EmailException, MalformedURLException {
180         sendComplex(new URL("https://www.apache.org/images/asf_logo_wide.gif"));
181     }
182 
183     /**
184      * Send HTML mail using the supplied inline image, without requiring a remote resource.
185      */
186     void sendComplex(URL imageUrl) throws EmailException {
187         htmlEmail.addTo(this.recipient.getEmail(), this.recipient.getFirstname() + " " + this.recipient.getSurname());
188         htmlEmail.setFrom(mailConfig.getFrom(), "Me");
189         htmlEmail.setSubject("HTML" + mailConfig.getSubject() + " " + new Date());
190 
191         // embed the image and get the content id
192         String cid = htmlEmail.embed(imageUrl, "ASF logo");
193 
194         // set the html message
195         htmlEmail.setHtmlMsg("<html>The ASF logo - <img src=\"cid:" + cid + "\"></html>");
196 
197         // set the alternative message
198         htmlEmail.setTextMsg("Your email client does not support HTML messages - thus no ASF logo visible.");
199 
200         if (mailConfig.sendOutMails()) {
201             htmlEmail.send();
202         }
203     }

43  /**
44   * Encapsulates the actual CSV parsing and mailing generation.
45   */
46  public class Mailversendala {
47      private static final Logger LOG =
48              LogManager.getLogger(Mailversendala.class);
49  
50      /**
51       * Send out mails with the given configuration and return result object afterwards.
52       * @param configuration application configuration.
53       * @return result object of the current application run.
54       * @throws IOException in case of errors.
55       */
56      public static MailingResult sendOut(MailConfig configuration) throws IOException {
57          LOG.info("**** MAILVERSENDALA: Starting .... ****");
58  
59          String csvPath = configuration.getCsvPath();
60          MailingResult result = new MailingResult();
61  
62          LOG.info("Consuming CSV: {}", csvPath);
63  
64          if (!Strings.isNullOrEmpty(csvPath)) {
65              File asFile = new File(configuration.getCsvPath());
66              if (asFile.exists()) {
67                  try (FileReader r = new FileReader(asFile, StandardCharsets.UTF_8)) {
68                      CsvParser parser = new CsvParser(r);
69                      final List<Mailing> mailings = parser.parse();
70  
71                      // Load and compile both external templates before sending any messages.
72                      final VelocityMailTemplate templates = new VelocityMailTemplate(configuration);
73                      String testAddress = configuration.getTo();
74                      boolean sendTestMail = testAddress != null && !testAddress.isBlank();
75                      final int total = mailings.size() + (sendTestMail ? 1 : 0);
76                      LOG.info("Will send out {} mails ... hold on tight :-)", total);
77  
78                      MailResources resources = new MailResources(configuration);
79                      List<PreparedMail> prepared = new ArrayList<>();
80                      if (sendTestMail) {
81                          // The single test mail deliberately keeps all placeholders unresolved.
82                          MailTemplate rawTemplate = new FileMailTemplate(configuration);
83                          resources.validate(rawTemplate);
84                          Mailing testRecipient = Mailing.builder().email(testAddress.trim()).build();
85                          prepared.add(new PreparedMail(testRecipient, rawTemplate));
86                          LOG.debug("Prepared unrendered test mail to: {}", testRecipient.email());
87                      }
88                      for (Mailing mailing : mailings) {
89                          try {
90                              MailTemplate rendered = templates.render(mailing);
91                              HtmlMarkupValidator.validate(rendered.getHtml(),
92                                      Path.of(configuration.getTemplatePath(), configuration.getHtmlTemplate())
93                                              + " (recipient: " + mailing.email() + ")");
94                              resources.validate(rendered);
95                              prepared.add(new PreparedMail(mailing, rendered));
96                          } catch (VelocityException e) {
97                              result.addError();
98                              LOG.error("Problem while rendering {}", mailing, e);
99                          }
100                     }
101                     // Validate every recipient's resource references before the first SMTP delivery.
102                     if (!prepared.isEmpty()) {
103                         LOG.info("SMTP transport: {}", configuration.getPort() == 465 ? "SMTPS" : "STARTTLS");
104                         if (!configuration.sendOutMails()) {
105                             LOG.warn("Running in DEMO mode, thus no mails will be sent out.");
106                         }
107                     }
108                     int mailNumber = 0;
109                     for (PreparedMail mail : prepared) {
110                         mailNumber++;
111                         try {
112                             new SendOut(mail.recipient(), configuration).send(mail.template(), resources);
113                             result.addSuccess();
114                             if (configuration.sendOutMails()) {
115                                 LOG.info("Mail {}/{} erfolgreich gesendet an: {}", mailNumber, total, mail.recipient().email());
116                             } else {
117                                 LOG.info("Mail {}/{} im DEMO-Modus vorbereitet für: {}", mailNumber, total, mail.recipient().email());
118                             }
119                         } catch (EmailException e) {
120                             result.addError();
121                             LOG.error("Problem while sending out {}", mail.recipient(), e);
122                         }
123                     }
124 
125                     LOG.info("**** MAILVERSENDALA-report: {} total mails ****", total);
126                     LOG.info("**** MAILVERSENDALA-report: {} successfully sent out ****", result.getMailCounter().orElse(new AtomicInteger(0)));
127                     LOG.info("**** MAILVERSENDALA-report: {} errors ****", result.getErrorCounter().orElse(new AtomicInteger(0)));
128                 }
129 
130             } else {
131                 LOG.warn("Nothing to do - please configure your CSV path properly, either as environment variable or as a runtime parameter. Example: java -Dcsvpath=foo -jar fatJar.jar");
132             }
133         }
134         LOG.info("**** MAILVERSENDALA: Application shutdown .... ****");
135         return result;
136     }
137 
138     private record PreparedMail(Mailing recipient, MailTemplate template) { }

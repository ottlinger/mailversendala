40  /**
41   * Encapsulates the actual CSV parsing and mailing generation.
42   */
43  public class Mailversendala {
44      private static final Logger LOG =
45              LogManager.getLogger(Mailversendala.class);
46  
47      /**
48       * Send out mails with the given configuration and return result object afterwards.
49       * @param configuration application configuration.
50       * @return result object of the current application run.
51       * @throws IOException in case of errors.
52       */
53      public static MailingResult sendOut(MailConfig configuration) throws IOException {
54          LOG.info("**** MAILVERSENDALA: Starting .... ****");
55  
56          String csvPath = configuration.getCsvPath();
57          MailingResult result = new MailingResult();
58  
59          LOG.info("Consuming CSV: {}", csvPath);
60  
61          if (!Strings.isNullOrEmpty(csvPath)) {
62              LOG.info(configuration.getCsvPath());
63  
64              File asFile = new File(configuration.getCsvPath());
65              if (asFile.exists()) {
66                  try (FileReader r = new FileReader(asFile, StandardCharsets.UTF_8)) {
67                      CsvParser parser = new CsvParser(r);
68                      final List<Mailing> mailings = parser.parse();
69  
70                      // Load and compile both external templates before sending any messages.
71                      final VelocityMailTemplate templates = new VelocityMailTemplate(configuration);
72                      final int total = mailings.size();
73                      LOG.info("Will send out {} mails ... hold on tight :-)", total);
74  
75                      MailResources resources = new MailResources(configuration);
76                      List<PreparedMail> prepared = new ArrayList<>();
77                      for (Mailing mailing : mailings) {
78                          try {
79                              MailTemplate rendered = templates.render(mailing);
80                              resources.validate(rendered);
81                              prepared.add(new PreparedMail(mailing, rendered));
82                          } catch (VelocityException e) {
83                              result.addError();
84                              LOG.error("Problem while rendering {}", mailing, e);
85                          }
86                      }
87                      // Validate every recipient's resource references before the first SMTP delivery.
88                      for (PreparedMail mail : prepared) {
89                          try {
90                              new SendOut(mail.recipient(), configuration).send(mail.template(), resources);
91                              result.addSuccess();
92                              LOG.info("Successfully send out {}.mail", result.getMailCounter().orElse(new AtomicInteger(0)));
93                          } catch (EmailException e) {
94                              result.addError();
95                              LOG.error("Problem while sending out {}", mail.recipient(), e);
96                          }
97                      }
98  
99                      LOG.info("**** MAILVERSENDALA-report: {} total mails ****", total);
100                     LOG.info("**** MAILVERSENDALA-report: {} successfully send out ****", result.getMailCounter().orElse(new AtomicInteger(0)));
101                     LOG.info("**** MAILVERSENDALA-report: {} errors ****", result.getErrorCounter().orElse(new AtomicInteger(0)));
102                 }
103 
104             } else {
105                 LOG.warn("Nothing to do - please configure your CSV path properly, either as environment variable or as a runtime parameter. Example: java -Dcsvpath=foo -jar fatJar.jar");
106             }
107         }
108         LOG.info("**** MAILVERSENDALA: Application shutdown .... ****");
109         return result;
110     }
111 
112     private record PreparedMail(Mailing recipient, MailTemplate template) { }
113 }

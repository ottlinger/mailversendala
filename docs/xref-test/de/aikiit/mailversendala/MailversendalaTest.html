45  @ExtendWith(MockitoExtension.class)
46  public class MailversendalaTest {
47  
48      private static final String CSV_NAME = "MailversendalaTest";
49  
50      @TempDir
51      public static Path testFolder;
52  
53      @Mock
54      private MailConfig configuration;
55  
56      @BeforeAll
57      public static void flushCSVData() throws IOException {
58          final Path csv = Files.createFile(testFolder.resolve(CSV_NAME));
59          Files.write(csv, (CsvParserTest.CSV_INPUT + "error,foo@bar.ru\n").getBytes());
60      }
61  
62      @Test
63      public void parseCSVFromARealFolder() throws IOException {
64          when(configuration.getCsvPath()).thenReturn(testFolder.toString() + File.separator + CSV_NAME);
65          when(configuration.getTemplatePath()).thenReturn("src/test/resources/template");
66          when(configuration.getHtmlTemplate()).thenReturn("mailtemplate.html");
67          when(configuration.getPlaintextTemplate()).thenReturn("mailtemplate.txt");
68          when(configuration.getHost()).thenReturn("localhost");
69          when(configuration.getPort()).thenReturn(465);
70          when(configuration.getUsername()).thenReturn("user");
71          when(configuration.getPassword()).thenReturn("password");
72          when(configuration.getFrom()).thenReturn("sender@example.test");
73          when(configuration.getSubject()).thenReturn("Template mailing");
74  
75          final MailingResult result = Mailversendala.sendOut(configuration);
76          assertThat(result).isNotNull();
77          assertThat(result.getErrorCounter()).isEmpty();
78  
79          final Optional<AtomicInteger> mailCounter = result.getMailCounter();
80          assertThat(mailCounter).isPresent();
81          assertThat(mailCounter.get().get()).isEqualTo(3);
82      }
83  
84      @Test
85      void missingTemplatesAbortBeforeAnySmtpClientIsCreated() {
86          when(configuration.getCsvPath()).thenReturn(testFolder.resolve(CSV_NAME).toString());
87          when(configuration.getTemplatePath()).thenReturn(testFolder.toString());
88          when(configuration.getHtmlTemplate()).thenReturn("missing.html");
89  
90          assertThatThrownBy(() -> Mailversendala.sendOut(configuration))
91                  .isInstanceOf(IOException.class).hasMessageContaining("Cannot load mail templates");
92          verify(configuration, never()).getHost();
93      }
94  
95      @Test
96      void missingImageForLaterRecipientAbortsBeforeFirstDelivery() throws IOException {
97          Path csv = testFolder.resolve("image-recipients.csv");
98          Files.writeString(csv, "firstname,surname,email\nvalid,One,first@example.test\nmissing,Two,second@example.test\n");
99          Files.writeString(testFolder.resolve("images.html"), "<p>Hello ${firstname}</p><img src='${firstname}.png'>");
100         Files.writeString(testFolder.resolve("images.txt"), "Hello ${firstname}");
101         try (var input = getClass().getResourceAsStream("/template/img/monkey-toy-16-118729.png")) {
102             Files.copy(input, testFolder.resolve("valid.png"));
103         }
104         when(configuration.getCsvPath()).thenReturn(csv.toString());
105         when(configuration.getTemplatePath()).thenReturn(testFolder.toString());
106         when(configuration.getHtmlTemplate()).thenReturn("images.html");
107         when(configuration.getPlaintextTemplate()).thenReturn("images.txt");
108 
109         assertThatThrownBy(() -> Mailversendala.sendOut(configuration))
110                 .isInstanceOf(IOException.class).hasMessageContaining("missing.png");
111         verify(configuration, never()).getHost();
112     }
113 
114     @Test
115     void countsRenderingErrorsWithoutAttemptingDelivery() throws IOException {
116         Files.writeString(testFolder.resolve("invalid-reference.html"), "<p>${unknownField}</p>");
117         Files.writeString(testFolder.resolve("valid.txt"), "Hello ${firstName}");
118         when(configuration.getCsvPath()).thenReturn(testFolder.resolve(CSV_NAME).toString());
119         when(configuration.getTemplatePath()).thenReturn(testFolder.toString());
120         when(configuration.getHtmlTemplate()).thenReturn("invalid-reference.html");
121         when(configuration.getPlaintextTemplate()).thenReturn("valid.txt");
122 
123         MailingResult result = Mailversendala.sendOut(configuration);
124 
125         assertThat(result.getTotal()).isEqualTo(3);
126         assertThat(result.getErrorCounter().orElseThrow().get()).isEqualTo(3);
127         assertThat(result.getMailCounter()).isEmpty();
128         verify(configuration, never()).getHost();
129     }

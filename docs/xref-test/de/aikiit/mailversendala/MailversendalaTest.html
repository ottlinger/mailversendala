99      @ParameterizedTest
100     @ValueSource(booleans = {false, true})
101     void sendsOneRawTestMailInNormalCsvRunEvenWhenReferencesCannotBeResolved(boolean unknownReference,
102                                                                              @TempDir Path folder) throws Exception {
103         Path csv = folder.resolve("recipients.csv");
104         Files.writeString(csv, "firstname,surname,email,salutation\n"
105                 + "Ada,Lovelace,ada@example.test,Dear\nGrace,Hopper,grace@example.test,Hello\n");
106         String reference = unknownReference ? "${unknownField}" : "${firstName}";
107         Files.writeString(folder.resolve("preview.html"), "<p>" + reference + "</p>");
108         Files.writeString(folder.resolve("preview.txt"), "Hello " + reference);
109         when(configuration.getCsvPath()).thenReturn(csv.toString());
110         when(configuration.getTemplatePath()).thenReturn(folder.toString());
111         when(configuration.getHtmlTemplate()).thenReturn("preview.html");
112         when(configuration.getPlaintextTemplate()).thenReturn("preview.txt");
113         when(configuration.getTo()).thenReturn("  myself@example.test  ");
114         List<Mailing> recipients = new ArrayList<>();
115 
116         try (MockedConstruction<SendOut> senders = mockConstruction(SendOut.class,
117                 (mock, context) -> recipients.add((Mailing) context.arguments().get(0)))) {
118             MailingResult result = Mailversendala.sendOut(configuration);
119 
120             assertThat(result.getTotal()).isEqualTo(3);
121             assertThat(result.getMailCounter().orElseThrow().get()).isEqualTo(unknownReference ? 1 : 3);
122             assertThat(result.getErrorCounter().map(AtomicInteger::get).orElse(0))
123                     .isEqualTo(unknownReference ? 2 : 0);
124             assertThat(recipients.get(0).email()).isEqualTo("myself@example.test");
125             ArgumentCaptor<MailTemplate> preview = ArgumentCaptor.forClass(MailTemplate.class);
126             verify(senders.constructed().get(0)).send(preview.capture(), any(MailResources.class));
127             assertThat(preview.getValue().getHtml()).isEqualTo("<p>" + reference + "</p>");
128             assertThat(preview.getValue().getPlaintext()).isEqualTo("Hello " + reference);
129             if (unknownReference) {
130                 assertThat(senders.constructed()).hasSize(1);
131             } else {
132                 assertThat(recipients).extracting(Mailing::email)
133                         .containsExactly("myself@example.test", "ada@example.test", "grace@example.test");
134                 ArgumentCaptor<MailTemplate> personalized = ArgumentCaptor.forClass(MailTemplate.class);
135                 verify(senders.constructed().get(1)).send(personalized.capture(), any(MailResources.class));
136                 assertThat(personalized.getValue().getHtml()).isEqualTo("<p>Ada</p>");
137             }
138         }
139     }

141     @ParameterizedTest
142     @ValueSource(strings = {"<spna>Bad</spna>", "<p>Bad</pp>"})
143     void invalidMarkupForLaterRecipientAbortsBeforeAnyDeliveryIncludingTestMail(String invalidHtml,
144                                                                                @TempDir Path folder) throws Exception {
145         Path csv = folder.resolve("recipients.csv");
146         Files.writeString(csv, "firstname,surname,email,salutation\n"
147                 + "Ada,Lovelace,ada@example.test,Dear\nGrace,Hopper,grace@example.test,Hello\n");
148         Files.writeString(folder.resolve("conditional.html"),
149                 "#if($firstName == 'Grace')\n" + invalidHtml + "\n#else\n<p>Hello ${firstName}</p>\n#end");
150         Files.writeString(folder.resolve("conditional.txt"), "Hello ${firstName}");
151         when(configuration.getCsvPath()).thenReturn(csv.toString());
152         when(configuration.getTemplatePath()).thenReturn(folder.toString());
153         when(configuration.getHtmlTemplate()).thenReturn("conditional.html");
154         when(configuration.getPlaintextTemplate()).thenReturn("conditional.txt");
155         when(configuration.getTo()).thenReturn("myself@example.test");
156 
157         try (MockedConstruction<SendOut> senders = mockConstruction(SendOut.class)) {
158             assertThatThrownBy(() -> Mailversendala.sendOut(configuration))
159                     .isInstanceOf(IOException.class)
160                     .hasMessageContaining("Invalid HTML markup")
161                     .hasMessageContaining("conditional.html")
162                     .hasMessageContaining("recipient: grace@example.test")
163                     .hasMessageContaining("line/column");
164             assertThat(senders.constructed()).isEmpty();
165         }
166         verify(configuration, never()).getHost();
167     }

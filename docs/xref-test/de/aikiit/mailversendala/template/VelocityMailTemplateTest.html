25  import java.io.IOException;
26  import java.io.StringReader;

28  import java.nio.file.Path;

30  import static org.assertj.core.api.Assertions.assertThat;
31  import static org.assertj.core.api.Assertions.assertThatThrownBy;
32  import static org.mockito.Mockito.mock;
33  import static org.mockito.Mockito.when;
34  
35  class VelocityMailTemplateTest {
36  
37      @TempDir
38      Path directory;
39  
40      @Test
41      void rendersExternalTemplatesWithEscapedHtmlAndUnescapedPlaintext() throws Exception {
42          VelocityMailTemplate templates = templates("<p>${firstName} ${lastName}: ${email}</p>",
43                  "Dear ${firstname} ${surname}: ${email}\n");
44          Mailing recipient = Mailing.builder().firstname("Jörg <Admin>").surname("Smith & Jones")
45                  .email("user@example.test").build();
46  
47          MailTemplate result = templates.render(recipient);
48  
49          assertThat(result.getHtml()).isEqualTo("<p>J&ouml;rg &lt;Admin&gt; Smith &amp; Jones: user@example.test</p>");
50          assertThat(result.getPlaintext())
51                  .isEqualTo("Dear Jörg <Admin> Smith & Jones: user@example.test\n");
52      }
53  
54      @Test
55      void compiledTemplatesKeepRecipientAndAlternativeContextsIsolated() throws Exception {
56          VelocityMailTemplate templates = templates("#set($firstName = 'HTML only')${firstName} ${surname}",
57                  "${firstName} ${lastName}");
58  
59          MailTemplate first = templates.render(Mailing.builder().firstname("Ada").surname("Lovelace").build());
60          MailTemplate second = templates.render(Mailing.builder().firstname("Grace").surname("Hopper").build());
61  
62          assertThat(first.getHtml()).isEqualTo("HTML only Lovelace");
63          assertThat(first.getPlaintext()).isEqualTo("Ada Lovelace");
64          assertThat(second.getHtml()).isEqualTo("HTML only Hopper");
65          assertThat(second.getPlaintext()).isEqualTo("Grace Hopper");
66      }
67  
68      @Test
69      void treatsAbsentRecipientFieldsAsEmptyStringsAndPreservesWhitespace() throws Exception {
70          VelocityMailTemplate templates = templates("  <p>${salutation}${firstname}${surname}${email}</p>\n",
71                  "\n${salutation}${firstName}${lastName}\n");
72  
73          MailTemplate result = templates.render(Mailing.builder().build());
74  
75          assertThat(result.getHtml()).isEqualTo("  <p></p>\n");
76          assertThat(result.getPlaintext()).isEqualTo("\n\n");
77      }
78  
79      @Test
80      void supportsVelocityConditionals() throws Exception {
81          VelocityMailTemplate templates = templates("#if($firstname)Hello ${firstname}#else Hello there#end",
82                  "#if($firstname)Hello ${firstname}#else Hello there#end");
83  
84          MailTemplate result = templates.render(Mailing.builder().firstname("Ada").build());
85  
86          assertThat(result.getHtml()).isEqualTo("Hello Ada");
87          assertThat(result.getPlaintext()).isEqualTo("Hello Ada");
88      }
89  
90      @Test
91      void rendersIndividualSalutationsFromCsvInBothAlternatives() throws Exception {
92          String csv = """
93                  firstname,surname,email,salutation
94                  Erika,Müller,erika@example.test,Liebe
95                  Peter,Schmidt,peter@example.test,Lieber
96                  """;
97          VelocityMailTemplate templates = templates("<p>${salutation} ${firstName}</p>",
98                  "${salutation} ${firstName}");
99          var recipients = new CsvParser(new StringReader(csv)).parse();
100 
101         MailTemplate erika = templates.render(recipients.get(0));
102         MailTemplate peter = templates.render(recipients.get(1));
103 
104         assertThat(erika.getHtml()).isEqualTo("<p>Liebe Erika</p>");
105         assertThat(erika.getPlaintext()).isEqualTo("Liebe Erika");
106         assertThat(peter.getHtml()).isEqualTo("<p>Lieber Peter</p>");
107         assertThat(peter.getPlaintext()).isEqualTo("Lieber Peter");
108     }
109 
110     @Test
111     void escapesSalutationOnlyInHtml() throws Exception {
112         VelocityMailTemplate templates = templates("<p>${salutation} ${firstName}</p>",
113                 "${salutation} ${firstName}");
114         Mailing recipient = Mailing.builder().salutation("Liebe <Freundin> & Kollegin").firstname("Erika").build();
115 
116         MailTemplate result = templates.render(recipient);
117 
118         assertThat(result.getHtml()).isEqualTo("<p>Liebe &lt;Freundin&gt; &amp; Kollegin Erika</p>");
119         assertThat(result.getPlaintext()).isEqualTo("Liebe <Freundin> & Kollegin Erika");
120     }
121 
122     @Test
123     void failsToLoadMissingTemplates() throws Exception {
124         MailConfig config = configuration();
125         Files.writeString(directory.resolve("custom.html"), "<p>Hello</p>");
126 
127         assertThatThrownBy(() -> new VelocityMailTemplate(config))
128                 .isInstanceOf(IOException.class).hasMessageContaining(directory.toString());
129     }
130 
131     @Test
132     void failsToLoadMalformedTemplates() {
133         assertThatThrownBy(() -> templates("#if($firstname)unterminated", "Hello"))
134                 .isInstanceOf(IOException.class);
135     }
136 
137     @Test
138     void rejectsUnknownPlaceholdersRatherThanSendingUnexpandedReferences() throws Exception {
139         VelocityMailTemplate templates = templates("<p>${unknownField}</p>", "Hello");
140 
141         assertThatThrownBy(() -> templates.render(Mailing.builder().build()))
142                 .isInstanceOf(VelocityException.class);
143     }
144 
145     private VelocityMailTemplate templates(String html, String plaintext) throws IOException {
146         Files.writeString(directory.resolve("custom.html"), html);
147         Files.writeString(directory.resolve("custom.txt"), plaintext);
148         return new VelocityMailTemplate(configuration());
149     }
150 
151     private MailConfig configuration() {
152         MailConfig config = mock(MailConfig.class);
153         when(config.getTemplatePath()).thenReturn(directory.toString());
154         when(config.getHtmlTemplate()).thenReturn("custom.html");
155         when(config.getPlaintextTemplate()).thenReturn("custom.txt");
156         return config;
157     }
158 }

52  /**
53   * Local mail delivery tests. GreenMail loads the test-only {@code /greenmail.p12}
54   * from the classpath, with its default password {@code changeit}. This fixture
55   * contains a self-signed localhost certificate and is not a production credential.
56   * To regenerate it from the project root:
57   * <pre>
58   * keytool -genkeypair -alias greenmail -keyalg RSA -keysize 2048 -storetype PKCS12
59   *   -keystore src/test/resources/greenmail.p12 -storepass changeit -keypass changeit
60   *   -dname "CN=localhost, OU=Tests, O=mailversendala"
61   *   -ext "SAN=dns:localhost,ip:127.0.0.1" -startdate "2020/01/01 00:00:00" -validity 36500
62   * </pre>
63   */

91          when(config.getFrom()).thenReturn(FROM);
92          when(config.getSubject()).thenReturn(SUBJECT);

103     @Test
104     void deliversPlainTextOverAuthenticatedSmtps() throws Exception {
105         newSendOut().send();
106 
107         MimeMessage message = receivedMessage();
108         assertAddressesAndHeaders(message);
109         assertThat(message.getSubject()).startsWith(SUBJECT + " ");
110         assertThat(message.isMimeType(EmailConstants.TEXT_PLAIN)).isTrue();
111         assertThat(message.getContentType()).containsIgnoringCase("charset=" + StandardCharsets.UTF_8.name());
112         assertThat(message.getContent().toString())
113                 .contains("This is a test mail from Mailversendala... :-)");
114     }

116     @Test
117     void deliversHtmlWithTextAlternativeAndInlineImage() throws Exception {
118         newSendOut().sendComplex(IMAGE);
119 
120         MimeMessage message = receivedMessage();
121         assertAddressesAndHeaders(message);
122         assertThat(message.getSubject()).startsWith("HTML" + SUBJECT + " ");
123         assertThat(((InternetAddress) message.getRecipients(Message.RecipientType.TO)[0]).getPersonal())
124                 .isEqualTo("Ada Lovelace");
125         assertThat(message.isMimeType("multipart/*")).isTrue();
126 
127         List<Part> parts = new ArrayList<>();
128         collectParts(message, parts);
129         Part text = onlyPartOfType(parts, EmailConstants.TEXT_PLAIN);
130         Part html = onlyPartOfType(parts, EmailConstants.TEXT_HTML);
131         Part image = onlyPartOfType(parts, "image/png");
132         assertThat(text.getContent().toString())
133                 .contains("Your email client does not support HTML messages - thus no ASF logo visible.");
134         String contentId = image.getHeader("Content-ID")[0];
135         assertThat(contentId).startsWith("<").endsWith(">");
136         assertThat(html.getContent().toString())
137                 .contains("cid:" + contentId.substring(1, contentId.length() - 1));
138         assertThat(image.getDisposition()).isEqualToIgnoringCase(Part.INLINE);
139         try (InputStream receivedImage = image.getInputStream(); InputStream fixture = IMAGE.openStream()) {
140             assertThat(receivedImage.readAllBytes()).isEqualTo(fixture.readAllBytes());
141         }
142     }

166     @Test
167     void refusesPlaintextFallbackWhenServerDoesNotSupportStartTls() throws Exception {
168         // GreenMail 1.6.x deliberately has no STARTTLS support on its SMTP listener.
169         GreenMail smtpOnly = new GreenMail(new ServerSetup(0, "127.0.0.1", ServerSetup.PROTOCOL_SMTP));
170         try {
171             smtpOnly.setUser(RECIPIENT, USERNAME, PASSWORD);
172             smtpOnly.start();
173             when(config.getPort()).thenReturn(smtpOnly.getSmtp().getPort());
174             SendOut sendOut = newSendOut();
175 
176             assertThatThrownBy(sendOut::send)
177                     .isInstanceOf(EmailException.class)
178                     .hasStackTraceContaining("STARTTLS is required but host does not support STARTTLS");
179             assertThat(smtpOnly.getReceivedMessages()).isEmpty();
180         } finally {
181             smtpOnly.stop();
182         }
183     }

185     @Test
186     void deliversExternalVelocityTemplatesAsUtf8Alternatives(@TempDir Path directory) throws Exception {
187         VelocityMailTemplate templates = externalTemplates(directory);
188         when(config.getSubject()).thenReturn("Einladung für Jörg");
189         Mailing recipient = Mailing.builder().email(RECIPIENT)
190                 .firstname("Jörg <Admin>").surname("Smith & Jones").build();
191 
192         newSendOut(recipient).send(templates.render(recipient));
193 
194         MimeMessage message = receivedMessage();
195         assertAddressesAndHeaders(message);
196         assertThat(message.getSubject()).isEqualTo("Einladung für Jörg");
197         List<Part> parts = new ArrayList<>();
198         collectParts(message, parts);
199         assertThat(parts).hasSize(2);
200         Part html = onlyPartOfType(parts, EmailConstants.TEXT_HTML);
201         Part text = onlyPartOfType(parts, EmailConstants.TEXT_PLAIN);
202         assertThat(html.getContentType()).containsIgnoringCase("charset=" + StandardCharsets.UTF_8.name());
203         assertThat(text.getContentType()).containsIgnoringCase("charset=" + StandardCharsets.UTF_8.name());
204         assertThat(html.getContent().toString())
205                 .contains("<p>Hello J&ouml;rg &lt;Admin&gt; Smith &amp; Jones</p>");
206         assertThat(text.getContent().toString()).contains("Hello Jörg <Admin> Smith & Jones");
207     }

209     @Test
210     void deliversSeparatelyPersonalizedMessagesFromCompiledTemplates(@TempDir Path directory) throws Exception {
211         VelocityMailTemplate templates = externalTemplates(directory);
212         Mailing second = Mailing.builder().email(RECIPIENT).firstname("Grace").surname("Hopper").build();
213 
214         newSendOut().send(templates.render(MAILING));
215         newSendOut(second).send(templates.render(second));
216 
217         assertThat(greenMail.waitForIncomingEmail(5000, 2)).isTrue();
218         MimeMessage[] messages = greenMail.getReceivedMessages();
219         assertThat(messages).hasSize(2);
220         List<Part> firstParts = new ArrayList<>();
221         List<Part> secondParts = new ArrayList<>();
222         collectParts(messages[0], firstParts);
223         collectParts(messages[1], secondParts);
224         assertThat(onlyPartOfType(firstParts, EmailConstants.TEXT_PLAIN).getContent().toString())
225                 .contains("Ada Lovelace").doesNotContain("Grace Hopper");
226         assertThat(onlyPartOfType(secondParts, EmailConstants.TEXT_HTML).getContent().toString())
227                 .contains("Grace Hopper").doesNotContain("Ada Lovelace");
228     }

240     private VelocityMailTemplate externalTemplates(Path directory) throws Exception {
241         Files.writeString(directory.resolve("welcome.html"), "<p>Hello ${firstName} ${lastName}</p>");
242         Files.writeString(directory.resolve("welcome.txt"), "Hello ${firstName} ${lastName}");
243         when(config.getTemplatePath()).thenReturn(directory.toString());
244         when(config.getHtmlTemplate()).thenReturn("welcome.html");
245         when(config.getPlaintextTemplate()).thenReturn("welcome.txt");
246         return new VelocityMailTemplate(config);
247     }

249     private SendOut newSendOut() throws EmailException {
250         return newSendOut(MAILING);
251     }
252 
253     private SendOut newSendOut(Mailing recipient) throws EmailException {
254         SimpleEmail text = new SimpleEmail();
255         HtmlEmail html = new HtmlEmail();
256         SendOut sendOut = new SendOut(recipient, config, text, html);
257         configureTestSession(text);
258         configureTestSession(html);
259         return sendOut;
260     }
261 
262     private void configureTestSession(Email email) throws EmailException {
263         if (email.isSSLOnConnect()) {
264             email.setSslSmtpPort(Integer.toString(greenMail.getSmtps().getPort()));
265         }
266         Properties properties = email.getMailSession().getProperties();
267         // Trust only this test server's self-signed certificate, in this session only.
268         // Production TLS settings and server identity verification are unchanged.
269         properties.setProperty("mail.smtp.ssl.trust", "localhost");
270         properties.setProperty(EmailConstants.MAIL_SMTP_CONNECTIONTIMEOUT, "5000");
271         properties.setProperty(EmailConstants.MAIL_SMTP_TIMEOUT, "5000");
272         properties.setProperty("mail.smtp.writetimeout", "5000");
273         if (!email.isSSLOnConnect()) {
274             assertThat(properties.getProperty(EmailConstants.MAIL_TRANSPORT_STARTTLS_ENABLE)).isEqualTo("true");
275             assertThat(properties.getProperty(EmailConstants.MAIL_TRANSPORT_STARTTLS_REQUIRED)).isEqualTo("true");
276         }
277         assertThat(properties.getProperty(EmailConstants.MAIL_SMTP_SSL_CHECKSERVERIDENTITY)).isEqualTo("true");
278     }

280     private MimeMessage receivedMessage() {
281         assertThat(greenMail.waitForIncomingEmail(5000, 1)).isTrue();
282         assertThat(greenMail.getReceivedMessages()).hasSize(1);
283         return greenMail.getReceivedMessages()[0];
284     }

286     private void assertAddressesAndHeaders(MimeMessage message) throws Exception {
287         assertThat(((InternetAddress) message.getFrom()[0]).getAddress()).isEqualTo(FROM);
288         assertThat(message.getAllRecipients()).hasSize(1);
289         assertThat(((InternetAddress) message.getAllRecipients()[0]).getAddress()).isEqualTo(RECIPIENT);
290         assertThat(message.getHeader("X-Mailer", null))
291                 .isEqualTo("Mailversendala - https://github.com/ottlinger/mailversendala");
292     }
293 
294     private void collectParts(Part part, List<Part> parts) throws Exception {
295         if (part.isMimeType("multipart/*")) {
296             Multipart multipart = (Multipart) part.getContent();
297             for (int i = 0; i < multipart.getCount(); i++) {
298                 collectParts(multipart.getBodyPart(i), parts);
299             }
300         } else {
301             parts.add(part);
302         }
303     }
304 
305     private Part onlyPartOfType(List<Part> parts, String mimeType) throws Exception {
306         List<Part> matching = new ArrayList<>();
307         for (Part part : parts) {
308             if (part.isMimeType(mimeType)) {
309                 matching.add(part);
310             }
311         }
312         assertThat(matching).hasSize(1);
313         return matching.get(0);
314     }

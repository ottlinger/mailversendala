57  /**
58   * Local mail delivery tests. GreenMail loads the test-only {@code /greenmail.p12}
59   * from the classpath, with its default password {@code changeit}. This fixture
60   * contains a self-signed localhost certificate and is not a production credential.
61   * To regenerate it from the project root:
62   * <pre>
63   * keytool -genkeypair -alias greenmail -keyalg RSA -keysize 2048 -storetype PKCS12
64   *   -keystore src/test/resources/greenmail.p12 -storepass changeit -keypass changeit
65   *   -dname "CN=localhost, OU=Tests, O=mailversendala"
66   *   -ext "SAN=dns:localhost,ip:127.0.0.1" -startdate "2020/01/01 00:00:00" -validity 36500
67   * </pre>
68   */

125     @Test
126     void deliversHtmlWithTextAlternativeAndInlineImage() throws Exception {
127         newSendOut().sendComplex(IMAGE);
128 
129         MimeMessage message = receivedMessage();
130         assertAddressesAndHeaders(message);
131         assertThat(message.getSubject()).startsWith("HTML" + SUBJECT + " ");
132         assertThat(((InternetAddress) message.getRecipients(Message.RecipientType.TO)[0]).getPersonal())
133                 .isEqualTo("Ada Lovelace");
134         assertThat(message.isMimeType("multipart/*")).isTrue();
135 
136         List<Part> parts = new ArrayList<>();
137         collectParts(message, parts);
138         Part text = onlyPartOfType(parts, EmailConstants.TEXT_PLAIN);
139         Part html = onlyPartOfType(parts, EmailConstants.TEXT_HTML);
140         Part image = onlyPartOfType(parts, IMAGE_PNG);
141         assertThat(text.getContent().toString())
142                 .contains("Your email client does not support HTML messages - thus no ASF logo visible.");
143         String contentId = image.getHeader("Content-ID")[0];
144         assertThat(contentId).startsWith("<").endsWith(">");
145         assertThat(html.getContent().toString())
146                 .contains("cid:" + contentId.substring(1, contentId.length() - 1));
147         assertThat(image.getDisposition()).isEqualToIgnoringCase(Part.INLINE);
148         try (InputStream receivedImage = image.getInputStream(); InputStream fixture = IMAGE.openStream()) {
149             assertThat(receivedImage.readAllBytes()).isEqualTo(fixture.readAllBytes());
150         }
151     }

195     void deliversExternalVelocityTemplatesAsUtf8Alternatives(@TempDir Path directory) throws Exception {
196         VelocityMailTemplate templates = externalTemplates(directory);
197         when(config.getSubject()).thenReturn("Einladung für Jörg");
198         Mailing recipient = Mailing.builder().email(RECIPIENT)

218     @Test
219     void deliversSeparatelyPersonalizedMessagesFromCompiledTemplates(@TempDir Path directory) throws Exception {
220         VelocityMailTemplate templates = externalTemplates(directory);
221         Mailing second = Mailing.builder().email(RECIPIENT).firstname("Grace").surname("Hopper").build();
222 
223         newSendOut().send(templates.render(MAILING));
224         newSendOut(second).send(templates.render(second));
225 
226         assertThat(greenMail.waitForIncomingEmail(5000, 2)).isTrue();
227         MimeMessage[] messages = greenMail.getReceivedMessages();
228         assertThat(messages).hasSize(2);
229         List<Part> firstParts = new ArrayList<>();
230         List<Part> secondParts = new ArrayList<>();
231         collectParts(messages[0], firstParts);
232         collectParts(messages[1], secondParts);
233         assertThat(onlyPartOfType(firstParts, EmailConstants.TEXT_PLAIN).getContent().toString())
234                 .contains("Ada Lovelace").doesNotContain("Grace Hopper");
235         assertThat(onlyPartOfType(secondParts, EmailConstants.TEXT_HTML).getContent().toString())
236                 .contains("Grace Hopper").doesNotContain("Ada Lovelace");
237     }

239     @Test
240     void deliversPersonalizedSalutationsFromCsv(@TempDir Path directory) throws Exception {
241         Files.writeString(directory.resolve("greeting.html"), "<p>${salutation} ${firstName}</p>");
242         Files.writeString(directory.resolve("greeting.txt"), "${salutation} ${firstName}");
243         when(config.getTemplatePath()).thenReturn(directory.toString());
244         when(config.getHtmlTemplate()).thenReturn("greeting.html");
245         when(config.getPlaintextTemplate()).thenReturn("greeting.txt");
246         VelocityMailTemplate templates = new VelocityMailTemplate(config);
247         String csv = "firstname,surname,email,salutation\n"
248                 + "Erika,Müller," + RECIPIENT + ",Liebe\n"
249                 + "Peter,Schmidt," + RECIPIENT + ",Lieber\n";
250 
251         for (Mailing recipient : new CsvParser(new StringReader(csv)).parse()) {
252             newSendOut(recipient).send(templates.render(recipient));
253         }
254 
255         assertThat(greenMail.waitForIncomingEmail(5000, 2)).isTrue();
256         MimeMessage[] messages = greenMail.getReceivedMessages();
257         assertThat(messages).hasSize(2);
258         String[] greetings = {"Liebe Erika", "Lieber Peter"};
259         for (int i = 0; i < messages.length; i++) {
260             List<Part> parts = new ArrayList<>();
261             collectParts(messages[i], parts);
262             assertThat(onlyPartOfType(parts, EmailConstants.TEXT_HTML).getContent().toString())
263                     .isEqualTo("<p>" + greetings[i] + "</p>");
264             assertThat(onlyPartOfType(parts, EmailConstants.TEXT_PLAIN).getContent().toString())
265                     .isEqualTo(greetings[i]);
266         }
267     }

279     @Test
280     void embedsLocalImagesAndDeliversPdfAndImageAttachments(@TempDir Path directory) throws Exception {
281         copyAssets(directory);
282         externalTemplates(directory);
283         Files.writeString(directory.resolve("with-images.html"),
284                 "<p>Hello ${firstName}</p><IMG SRC=images/logo.png><img src='./images/logo.png'>");
285         when(config.getHtmlTemplate()).thenReturn("with-images.html");
286         when(config.getAttachments()).thenReturn("attachments/information.pdf, images/logo.png, attachments/information.pdf");
287         VelocityMailTemplate templates = new VelocityMailTemplate(config);
288 
289         newSendOut().send(templates.render(MAILING));
290 
291         MimeMessage message = receivedMessage();
292         assertAddressesAndHeaders(message);
293         assertThat(message.isMimeType(MULTIPART_MIXED)).isTrue();
294         List<Part> parts = new ArrayList<>();
295         collectParts(message, parts);
296         assertThat(parts).hasSize(5);
297         Part pdf = onlyPartOfType(parts, APPLICATION_PDF);
298         assertThat(pdf.getDisposition()).isEqualToIgnoringCase(Part.ATTACHMENT);
299         assertThat(pdf.getFileName()).isEqualTo("information.pdf");
300         assertPartBytes(pdf, Files.readAllBytes(directory.resolve("attachments/information.pdf")));
301         Part inline = null;
302         Part attachedImage = null;
303         for (Part part : parts) {
304             if (part.isMimeType(IMAGE_PNG)) {
305                 if (Part.INLINE.equalsIgnoreCase(part.getDisposition())) {
306                     inline = part;
307                 } else {
308                     attachedImage = part;
309                 }
310             }
311         }
312         assertThat(inline).isNotNull();
313         assertThat(attachedImage).isNotNull();
314         assertThat(attachedImage.getDisposition()).isEqualToIgnoringCase(Part.ATTACHMENT);
315         assertThat(attachedImage.getFileName()).isEqualTo("logo.png");
316         byte[] imageBytes = Files.readAllBytes(directory.resolve("images/logo.png"));
317         assertPartBytes(inline, imageBytes);
318         assertPartBytes(attachedImage, imageBytes);
319         String header = inline.getHeader("Content-ID")[0];
320         String cid = "cid:" + header.substring(1, header.length() - 1);
321         assertThat(Jsoup.parse(onlyPartOfType(parts, EmailConstants.TEXT_HTML).getContent().toString()).select("img"))
322                 .extracting(image -> image.attr("src")).containsExactly(cid, cid);
323         assertThat(onlyPartOfType(parts, EmailConstants.TEXT_PLAIN).getContent().toString()).contains("Hello Ada Lovelace");
324     }

388     private SendOut newSendOut(Mailing recipient) throws EmailException {
389         SimpleEmail text = new SimpleEmail();
390         HtmlEmail html = new HtmlEmail();
391         SendOut sendOut = new SendOut(recipient, config, text, html);
392         configureTestSession(text);
393         configureTestSession(html);
394         return sendOut;
395     }

397     private void configureTestSession(Email email) throws EmailException {
398         if (email.isSSLOnConnect()) {
399             email.setSslSmtpPort(Integer.toString(greenMail.getSmtps().getPort()));
400         }
401         Properties properties = email.getMailSession().getProperties();
402         // Trust only this test server's self-signed certificate, in this session only.
403         // Production TLS settings and server identity verification are unchanged.
404         properties.setProperty("mail.smtp.ssl.trust", "localhost");
405         properties.setProperty(EmailConstants.MAIL_SMTP_CONNECTIONTIMEOUT, "5000");
406         properties.setProperty(EmailConstants.MAIL_SMTP_TIMEOUT, "5000");
407         properties.setProperty("mail.smtp.writetimeout", "5000");
408         if (!email.isSSLOnConnect()) {
409             assertThat(properties.getProperty(EmailConstants.MAIL_TRANSPORT_STARTTLS_ENABLE)).isEqualTo("true");
410             assertThat(properties.getProperty(EmailConstants.MAIL_TRANSPORT_STARTTLS_REQUIRED)).isEqualTo("true");
411         }
412         assertThat(properties.getProperty(EmailConstants.MAIL_SMTP_SSL_CHECKSERVERIDENTITY)).isEqualTo("true");
413     }

421     private void assertAddressesAndHeaders(MimeMessage message) throws Exception {
422         assertThat(((InternetAddress) message.getFrom()[0]).getAddress()).isEqualTo(FROM);
423         assertThat(message.getAllRecipients()).hasSize(1);
424         assertThat(((InternetAddress) message.getAllRecipients()[0]).getAddress()).isEqualTo(RECIPIENT);
425         assertThat(message.getHeader("X-Mailer", null))
426                 .isEqualTo("Mailversendala - https://github.com/ottlinger/mailversendala");
427     }

429     private void collectParts(Part part, List<Part> parts) throws Exception {
430         if (part.isMimeType("multipart/*")) {
431             Multipart multipart = (Multipart) part.getContent();
432             for (int i = 0; i < multipart.getCount(); i++) {
433                 collectParts(multipart.getBodyPart(i), parts);
434             }
435         } else {
436             parts.add(part);
437         }
438     }
439 
440     private Part onlyPartOfType(List<Part> parts, String mimeType) throws Exception {
441         List<Part> matching = new ArrayList<>();
442         for (Part part : parts) {
443             if (part.isMimeType(mimeType)) {
444                 matching.add(part);
445             }
446         }
447         assertThat(matching).hasSize(1);
448         return matching.get(0);
449     }
